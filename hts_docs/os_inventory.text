-- =========================
-- ROLES
-- =========================
CREATE TABLE tbl_roles (
    RoleId INT AUTO_INCREMENT PRIMARY KEY,
    RoleName VARCHAR(50) NOT NULL
);

-- =========================
-- USERS
-- =========================
CREATE TABLE tbl_users (
    UserId INT AUTO_INCREMENT PRIMARY KEY,
    FullName VARCHAR(100) NOT NULL,
    Password VARCHAR(255) NOT NULL,
    Email VARCHAR(255) NOT NULL UNIQUE,
    RoleId INT DEFAULT 2,
    FailedAttempts INT DEFAULT 0,
    LockoutEnd DATETIME NULL,
    FOREIGN KEY (RoleId) REFERENCES tbl_roles(RoleId)
);

-- =========================
-- ASSETS
-- =========================
CREATE TABLE tbl_assets (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    CollCode VARCHAR(100),
    Name VARCHAR(255),
    Desc1 TEXT,
    AnDate DATE,
    MrNum VARCHAR(100),
    PropNo VARCHAR(100),
    Qty INT,
    UM VARCHAR(50),
    UCost DECIMAL(18,2),
    TCost DECIMAL(18,2),
    UserName VARCHAR(100),
    Email VARCHAR(100),
    CurrentUser VARCHAR(100),
    EqStatus VARCHAR(100),
    SerialNumber VARCHAR(100),
    Location VARCHAR(100),
    UserId INT,
    FOREIGN KEY (UserId) REFERENCES tbl_users(UserId)
);

-- =========================
-- TICKETS
-- =========================
CREATE TABLE tbl_tickets (
    TicketId INT AUTO_INCREMENT PRIMARY KEY,
    TicketNumber VARCHAR(255),
    PropNo VARCHAR(255),
    FullName VARCHAR(255),
    Email VARCHAR(255),
    RequestDate DATETIME,
    HelpTopic VARCHAR(100),
    IssueDesc VARCHAR(200),
    Location VARCHAR(255),
    PriorityLevel VARCHAR(50),
    CurrentStatus VARCHAR(100),
    LastUpdated DATETIME,
    DueDate DATETIME NULL,
    Overdue TINYINT(1),
    AgentAssigned VARCHAR(255) NULL,
    ReopenCount INT DEFAULT 0,

    -- NEW (non-breaking)
    UserId INT NULL,

    FileName VARCHAR(255) NULL,
    FileContentType VARCHAR(50) NULL,
    ExtraFields JSON NULL,

    FOREIGN KEY (UserId) REFERENCES tbl_users(UserId)
);

-- =========================
-- TICKET MESSAGES
-- =========================
CREATE TABLE tbl_ticketmessages (
    MessageId INT AUTO_INCREMENT PRIMARY KEY,
    TicketId INT NOT NULL,
    SenderEmail VARCHAR(255) NOT NULL,
    Message TEXT NOT NULL,
    Timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,

    -- NEW (optional relationship)
    UserId INT NULL,

    FileName VARCHAR(255) NULL,
    FilePath VARCHAR(500) NULL,
    FileType VARCHAR(100) NULL,
    FileSize BIGINT NULL,

    FOREIGN KEY (TicketId) REFERENCES tbl_tickets(TicketId),
    FOREIGN KEY (UserId) REFERENCES tbl_users(UserId)
);

-- =========================
-- TICKET FEEDBACK
-- =========================
CREATE TABLE tbl_ticketfeedback (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    TicketNumber VARCHAR(50) NOT NULL,

    -- NEW (non-breaking relationship)
    TicketId INT NULL,

    CompanyName VARCHAR(100),
    Unit VARCHAR(50),
    FullName VARCHAR(100),
    Email VARCHAR(100),
    Phone VARCHAR(20),
    SupportAgent VARCHAR(100),
    IssueType VARCHAR(255),
    SupportChannel VARCHAR(255),
    OtherIssue VARCHAR(255),
    DateRequested DATE,
    DateResolved DATE,
    FeedbackText TEXT,
    Rating INT,
    ResponseTime VARCHAR(20),
    TechnicalKnowledge VARCHAR(20),
    Professionalism VARCHAR(20),
    Communication VARCHAR(20),
    Resolution VARCHAR(20),
    Signature VARCHAR(255),
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (TicketId) REFERENCES tbl_tickets(TicketId)
);

-- =========================
-- TICKET NOTIFICATIONS
-- =========================
CREATE TABLE tbl_ticketnotifications (
    NotificationId INT AUTO_INCREMENT PRIMARY KEY,
    TicketId INT NOT NULL,
    RecipientEmail VARCHAR(255) NOT NULL,
    SenderEmail VARCHAR(255) NOT NULL,
    Message TEXT,
    Timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
    IsRead BOOLEAN DEFAULT FALSE,

    FOREIGN KEY (TicketId) REFERENCES tbl_tickets(TicketId)
);

-- =========================
-- ZOOM SCHEDULE
-- =========================
CREATE TABLE tbl_zoom_schedule (

    OfficeUnitProject VARCHAR(150),
    AdditionalDetails TEXT,
    Score INT,
    EmailAddress VARCHAR(150),
    ConfigFile VARCHAR(255),
    DateAddressed DATE,
    Status VARCHAR(50)
);
